################ MASALAH #1: TIMEZONE FILTER TANGGAL DI LAPORAN ################

PRIORITAS  : SANGAT KRITIS
FILE TERKAIT:
  - admin-public/laporan.js (fungsi getReportRange)
  - index.js (endpoint /api/admin/report/summary)

GEJALA:
  Ketika user pilih filter "Hari Ini" di menu Laporan, range tanggal yang
  muncul = KEMARIN (bukan hari ini).
  
  Contoh: Hari ini 12/10/2026 jam 08:00 WIB, tapi range muncul:
  "11/10/2026 - 11/10/2026"  SALAH

AKAR MASALAH:
  Kode pakai `new Date().toISOString().slice(0, 10)` yang menghasilkan
  tanggal UTC. Karena WIB = UTC+7, kalau jam < 07:00 pagi maka
  toISOString() menghasilkan tanggal kemarin.

  Contoh detail:
    Waktu WIB : 12/10/2026 00:27
    Waktu UTC : 11/10/2026 17:27
    toISOString() menghasilkan "2026-10-11" SALAH (harusnya "2026-10-12")

LOKASI KODE LAMA:

  #1a. File: admin-public/laporan.js
  ------------------------------------------------------------
  function getReportRange() {
    const preset = document.getElementById('reportRange')?.value || '30d';
    const now = new Date();
    let from;

    if (preset === 'today') {
      from = new Date(now); from.setHours(0, 0, 0, 0);
    } else if (preset === '7d') {
      from = new Date(now.getTime() - 7 * 86400000);
    } else if (preset === '30d') {
      from = new Date(now.getTime() - 30 * 86400000);
    } else if (preset === '90d') {
      from = new Date(now.getTime() - 90 * 86400000);
    } else if (preset === '1y') {
      from = new Date(now.getTime() - 365 * 86400000);
    } else if (preset === 'custom') {
      const fromVal = document.getElementById('reportFrom')?.value;
      const toVal = document.getElementById('reportTo')?.value;
      return {
        from: fromVal || new Date(now.getTime() - 30 * 86400000).toISOString().slice(0, 10),
        to: toVal || now.toISOString().slice(0, 10)
      };
    } else {
      from = new Date(now.getTime() - 30 * 86400000);
    }

    return {
      from: from.toISOString().slice(0, 10),
      to: now.toISOString().slice(0, 10)
    };
  }
  ------------------------------------------------------------

  #1b. File: index.js (endpoint /api/admin/report/summary)
  ------------------------------------------------------------
  const { from, to } = req.query;

  const now = new Date();
  const startDate = from ? new Date(from) : new Date(now.getTime() - 30 * 24 * 60 * 60 * 1000);
  const endDate = to ? new Date(to) : now;
  endDate.setHours(23, 59, 59, 999);

  const startMs = startDate.getTime();
  const endMs = endDate.getTime();
  ------------------------------------------------------------

SOLUSI YANG DIBUTUHKAN:
  1. Buat helper `toLocalDateString(date)` yang pakai metode LOCAL:
     - getFullYear()
     - getMonth()
     - getDate()
  
  2. Di laporan.js: ganti `toISOString().slice(0, 10)` dengan
     `toLocalDateString(date)`
  
  3. Di index.js: parse `from` dan `to` sebagai LOCAL timezone:
     const [y, m, d] = from.split('-').map(Number);
     const startDate = new Date(y, m - 1, d, 0, 0, 0, 0);

YANG SAYA INGINKAN SETELAH PERBAIKAN:
  - Filter "Hari Ini" tampil: 12/10/2026 - 12/10/2026
  - Filter "7 Hari" tampil : 06/10/2026 - 12/10/2026 (H-6 sampai H)
  - Filter "30 Hari" tampil: 13/09/2026 - 12/10/2026 (H-29 sampai H)

LOKASI KODE LAMA:

  File: lib/feedback.js
  ------------------------------------------------------------
  export function addFeedback(question, answer, rating, sessionId, meta = {}) {
      const item = {
          id: 'fb_' + Date.now() + '_' + Math.random().toString(36).slice(2, 6),
          question: question.slice(0, 500),
          answer: answer.slice(0, 1000),
          rating: rating,
          sessionId: sessionId || 'unknown',
          provider: meta.provider || 'unknown',
          model: meta.model || 'unknown',
          source: meta.source || 'ai',
          timestamp: new Date().toISOString()
      };
      feedbackData.push(item);

      const day = new Date().toISOString().slice(0, 10);
      if (!analyticsData.dailyStats[day]) {
          analyticsData.dailyStats[day] = { chats: 0, good: 0, bad: 0 };
      }
      if (rating === 'good') analyticsData.dailyStats[day].good++;
      else if (rating === 'bad') analyticsData.dailyStats[day].bad++;

      feedbackSave();
      analyticsSave();
      return item;
  }

  export function trackEvent(type) {
      analyticsData.totalChats++;
      if (type === 'refusal') analyticsData.totalRefusals++;
      if (type === 'memory') analyticsData.totalMemoryHits++;
      if (type === 'rag') analyticsData.totalRagHits++;
      if (type === 'out_of_scope') analyticsData.totalOutOfScope++;
      if (type === 'answer') analyticsData.totalAnswers++;

      const day = new Date().toISOString().slice(0, 10);
      if (!analyticsData.dailyStats[day]) {
          analyticsData.dailyStats[day] = { chats: 0, good: 0, bad: 0 };
      }
      analyticsData.dailyStats[day].chats++;

      analyticsSave();
  }
  ------------------------------------------------------------

SOLUSI YANG DIBUTUHKAN:
  1. Tambah helper `getLocalDateKey()` di atas file:
     function getLocalDateKey(date = new Date()) {
         const y = date.getFullYear();
         const m = String(date.getMonth() + 1).padStart(2, '0');
         const d = String(date.getDate()).padStart(2, '0');
         return `${y}-${m}-${d}`;
     }
  
  2. Ganti semua `new Date().toISOString().slice(0, 10)` di file ini
     dengan `getLocalDateKey()`

YANG SAYA INGINKAN SETELAH PERBAIKAN:

  ------------------------------------------------------------
  function renderFeedbackTrend(daily, feedback) {
    const el = document.getElementById('feedbackTrendChart');
    if (!el) return;
    if (!daily || daily.length === 0) {
      el.innerHTML = '<div class="text-center text-secondary py-4">Belum ada data</div>';
      return;
    }
    el.innerHTML = '<canvas></canvas>';
    const ctx = el.querySelector('canvas').getContext('2d');
    const good = Math.floor((feedback?.good || 0) / daily.length);
    const bad = Math.floor((feedback?.bad || 0) / daily.length);
    if (feedbackTrendChartInstance) feedbackTrendChartInstance.destroy();
    feedbackTrendChartInstance = new Chart(ctx, {
      type: 'line',
      data: {
        labels: daily.map(d => d.day),
        datasets: [
          {
            label: 'Bagus',
            data: daily.map(() => good),
          },
          {
            label: 'Kurang',
            data: daily.map(() => bad),
          }
        ]
      }
    });
  }
  ------------------------------------------------------------
